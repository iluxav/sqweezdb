=== E16: access control (roles, grants, row policies) ===
--- behaviour ---
app:    SELECT count(*) FROM docs                                      -> 1 row(s) [Int(1000000)]
root:   SELECT count(*) FROM docs                                      -> 1 row(s) [Int(1000000)]
alice:  SELECT count(*) FROM docs            (policy: tenant = 7)      -> 1 row(s) [Int(1000)]
alice:  SELECT title FROM docs WHERE id = 7  (tenant 7)                -> 1 row(s) [Text("doc 7")]
alice:  SELECT title FROM docs WHERE id = 8  (tenant 8)                -> 0 row(s)
alice:  SELECT count(*) FROM docs WHERE tenant = 8 (asks for another tenant) -> 1 row(s) [Int(0)]
alice:  INSERT a doc with tenant 8                                     -> ERROR: permission denied: row violates a policy on docs
alice:  INSERT a doc with tenant 7                                     -> 0 row(s)
alice:  UPDATE docs SET title = 'x' WHERE id = 7  (no UPDATE grant)    -> ERROR: permission denied: user alice lacks UPDATE on docs
bob:    SELECT count(*) FROM docs            (policy: tenant = 8)      -> 1 row(s) [Int(1000)]
nobody: SELECT count(*) FROM docs            (no grants)               -> ERROR: permission denied: user nobody lacks SELECT on docs
alice:  CREATE ROLE hacker                                             -> ERROR: permission denied: only the application session can manage roles, users and policies
root:   UPDATE docs SET title = 'x' WHERE id = 7                       -> 0 row(s)
alice:  SELECT title FROM docs WHERE id = 7  (sees root's change)      -> 1 row(s) [Text("x")]
